#include <LiquidCrystal.h>
// LCD Connections: RS=12, E=11, D4=5, D5=4, D6=3, D7=2LiquidCrystal lcd(12, 11, 5, 4, 3, 2);
const int LDR_PIN = A0;const float CONSUMPTION_THRESHOLD = 90.0; // Strictly > 90.0% is required for safe drinking water
void setup() {
  // -------------------------------------------------------------
  // 1. SETUP 40kHz PWM ON TIMER1 (PINS D9 & D10)
  // -------------------------------------------------------------
  pinMode(9, OUTPUT);    // OC1A - OUT1 driver signal
  pinMode(10, OUTPUT);   // OC1B - OUT2 driver signal
  pinMode(8, OUTPUT);    // ENA  - L298N Channel A Enable
  digitalWrite(8, HIGH); // Drive ENA HIGH to enable the L298N

  TCCR1A = 0;
  TCCR1B = 0;
  TCNT1  = 0;

  TCCR1A = (1 << COM1A1) | (1 << COM1B1) | (1 << COM1B0); 
  TCCR1B = (1 << WGM13) | (1 << CS10);                    
  ICR1   = 200;                                           // 40kHz signal
  OCR1A  = 100;                                           // 50% duty cycle
  OCR1B  = 100;

  // -------------------------------------------------------------
  // 2. SETUP LCD DISPLAY
  // -------------------------------------------------------------
  lcd.begin(16, 2);
  lcd.print("Water Tester");
  delay(1000);
  lcd.clear();
}
void loop() {
  int rawADC = analogRead(LDR_PIN);

  // Inverted Mapping:
  // As light INCREASES, LDR resistance drops, rawADC drops near 0.
  // (1023.0 - rawADC) converts 0 ADC -> 100.0% Clarity.
  float clarityPercent = ((1023.0 - rawADC) / 1023.0) * 100.0;
  
  // Constrain range safely between 0.0% and 100.0%
  if (clarityPercent > 100.0) clarityPercent = 100.0;
  if (clarityPercent < 0.0)   clarityPercent = 0.0;

  // Line 1: Display Percentage
  lcd.setCursor(0, 0);
  lcd.print("Clarity: ");
  lcd.print(clarityPercent, 1);
  lcd.print("%   ");

  // Line 2: Evaluate > 90.0% threshold
